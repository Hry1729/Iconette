const {useState,useEffect,useRef}=React;
const {NavLink,TextInput,Button,Avatar,Paper,Text,Tabs,Table,Badge,Checkbox,Select,Pagination,Menu,SegmentedControl,ActionIcon,Modal,Group,MantineProvider,ThemeIcon,Burger}=M;
const I=({n,className})=>html`<i className=${'ico '+(className||'')} style=${ic(n)}></i>`;
function Dash(){
 const [nav,setNav]=useState('仪表盘');const [range,setRange]=useState('90');const [tab,setTab]=useState('outline');const [rows,setRows]=useState(DOCS);const [sel,setSel]=useState([]);
 const [vis,setVis]=useState({type:true,status:true,target:true,limit:true,reviewer:true});const [ps,setPs]=useState('10');const [pg,setPg]=useState(1);const [o,setO]=useState(false);const [nt,setNt]=useState('');
 const chart=useRef();useEffect(()=>drawChart(chart.current,'#228be6',+range),[range,tab]);
 const n=+ps,pages=Math.ceil(rows.length/n),view=rows.slice((pg-1)*n,pg*n),ids=view.map(r=>r.id),all=ids.length&&ids.every(i=>sel.includes(i)),some=ids.some(i=>sel.includes(i));
 const notify=(m,c)=>notifications.show({message:m,color:c||'blue'});
 const navI=([i,l])=>html`<${NavLink} key=${l} label=${l} leftSection=${html`<${I} n=${i}/>`} active=${nav===l} onClick=${()=>setNav(l)} style=${{borderRadius:8}} py=${7}/>`;
 const create=()=>{setRows(rs=>[{id:Date.now(),header:nt||'未命名章节',type:'叙述',status:'proc',target:0,limit:0,reviewer:null},...rs]);setNt('');setPg(1);setO(false);notify('章节已创建','teal')};
 return html`<${React.Fragment}><div className="d2"><aside className="sb"><div className="sbh"><${ThemeIcon} size=${26} radius="md"><${I} n="circle-dashed"/><//>Acme Inc.</div>
  <div className="gl">首页</div>${NAV1.map(navI)}<div className="gl">文档</div>${NAV2.map(navI)}
  <div className="sbf">${NAV3.map(navI)}<${Menu} position="right-end" offset=${8} width=${180}><${Menu.Target}><${Paper} className="user" withBorder style=${{cursor:'pointer'}}><${Avatar} radius="md" color="blue">MT<//><div style=${{flex:1,minWidth:0}}><${Text} size="sm" fw=${500}>mantine<//><${Text} size="xs" c="dimmed">m@example.com<//></div><${I} n="chevrons-up-down"/><//><//><${Menu.Dropdown}><${Menu.Label}>我的账户<//>${[['user','个人资料'],['credit-card','账单'],['bell','通知']].map(([i,l])=>html`<${Menu.Item} key=${l} leftSection=${html`<${I} n=${i}/>`}>${l}<//>`)}<${Menu.Divider}/><${Menu.Item} color="red" leftSection=${html`<${I} n="log-out"/>`}>退出登录<//><//><//></div></aside>
 <div className="mn"><header className="top"><${Burger} size="sm" opened=${false}/><b>${nav}</b><span style=${{flex:1}}></span><${Button} size="xs" leftSection=${html`<${I} n="circle-plus"/>`} onClick=${()=>setO(true)}>快速创建<//></header>
 <div className="body">
  <div className="k4">${KPI.map(([t,v,d,up,f1,f2])=>html`<${Paper} key=${t} withBorder radius="md" p="lg" shadow="xs"><div className="kh"><span>${t}</span><${Badge} variant="light" color=${up?'teal':'red'} radius="sm" leftSection=${html`<${I} n=${up?'trending-up':'trending-down'} className="s12"/>`}>${d}<//></div><div className="kv">${v}</div><div className="kf"><b>${f1}</b><small>${f2}</small></div><//>`)}</div>
  <${Paper} withBorder radius="md" p="lg" shadow="xs"><div className="chh"><div><${Text} fw=${600}>访客总数<//><${Text} size="sm" c="dimmed">${RL[range]}总计<//></div><${SegmentedControl} size="xs" radius="md" value=${range} onChange=${setRange} data=${RLA.map(([value,label])=>({value,label}))}/></div><div className="chart" ref=${chart}></div><//>
  <div className="tb"><${Tabs} value=${tab} onChange=${setTab} variant="pills" radius="md"><${Tabs.List}>${TABS.map(([v,l,c])=>html`<${Tabs.Tab} key=${v} value=${v} rightSection=${c?html`<${Badge} size="xs" circle variant=${tab===v?'white':'light'}>${c}<//>`:null}>${l}<//>`)}<//><//>
   <${Group} gap="xs"><${Menu} closeOnItemClick=${false} position="bottom-end" width=${170}><${Menu.Target}><${Button} size="xs" variant="default" leftSection=${html`<${I} n="columns-2"/>`} rightSection=${html`<${I} n="chevron-down"/>`}>自定义列<//><//><${Menu.Dropdown}>${COLS.map(([k,l])=>html`<${Menu.Item} key=${k} onClick=${()=>setVis(x=>({...x,[k]:!x[k]}))} leftSection=${html`<${Checkbox} size="xs" checked=${vis[k]} readOnly/>`}>${l}<//>`)}<//><//><${Button} size="xs" variant="default" leftSection=${html`<${I} n="plus"/>`} onClick=${()=>setO(true)}>添加章节<//><//></div>
  ${tab!=='outline'?html`<div className="empty2">暂无内容</div>`:html`<${React.Fragment}><${Paper} withBorder radius="md" style=${{overflow:'hidden'}}><${Table} highlightOnHover verticalSpacing=${7} fz="sm"><${Table.Thead} bg="gray.0"><${Table.Tr}><${Table.Th} w=${30}/><${Table.Th} w=${34}><${Checkbox} size="xs" checked=${!!all} indeterminate=${some&&!all} onChange=${()=>setSel(s=>all?s.filter(i=>!ids.includes(i)):[...new Set([...s,...ids])])}/><//><${Table.Th}>标题<//>
   ${vis.type?html`<${Table.Th}>章节类型<//>`:null}${vis.status?html`<${Table.Th}>状态<//>`:null}${vis.target?html`<${Table.Th} ta="right">目标<//>`:null}${vis.limit?html`<${Table.Th} ta="right">上限<//>`:null}${vis.reviewer?html`<${Table.Th} pl=${24}>审核人<//>`:null}<${Table.Th} w=${44}/><//><//>
   <${Table.Tbody}>${view.map(r=>html`<${Table.Tr} key=${r.id} bg=${sel.includes(r.id)?'var(--mantine-color-blue-light)':undefined}><${Table.Td} c="dimmed"><${I} n="grip-vertical" className="grip"/><//><${Table.Td}><${Checkbox} size="xs" checked=${sel.includes(r.id)} onChange=${()=>setSel(s=>s.includes(r.id)?s.filter(i=>i!==r.id):[...s,r.id])}/><//><${Table.Td} fw=${500}>${r.header}<//>
    ${vis.type?html`<${Table.Td}><${Badge} variant="outline" color="gray" radius="sm" tt="none" fw=${500}>${r.type}<//><//>`:null}
    ${vis.status?html`<${Table.Td}><${Badge} variant="light" color=${r.status==='done'?'teal':'yellow'} radius="sm" tt="none" fw=${500} leftSection=${html`<${I} n=${r.status==='done'?'mdi/check-circle':'loader'} className="s12"/>`}>${r.status==='done'?'已完成':'进行中'}<//><//>`:null}
    ${vis.target?html`<${Table.Td} ta="right">${r.target}<//>`:null}${vis.limit?html`<${Table.Td} ta="right">${r.limit}<//>`:null}
    ${vis.reviewer?html`<${Table.Td} pl=${24}>${r.reviewer||html`<${Select} size="xs" w=${140} placeholder="分配审核人" data=${REV} onChange=${v=>setRows(rs=>rs.map(x=>x.id===r.id?{...x,reviewer:v}:x))}/>`}<//>`:null}
    <${Table.Td}><${Menu} position="bottom-end" width=${130}><${Menu.Target}><${ActionIcon} variant="subtle" color="gray" size="sm"><${I} n="ellipsis-vertical"/><//><//><${Menu.Dropdown}>${['编辑','复制','收藏'].map(x=>html`<${Menu.Item} key=${x} onClick=${()=>notify(x+'：'+r.header)}>${x}<//>`)}<${Menu.Divider}/><${Menu.Item} color="red" onClick=${()=>{setRows(rs=>rs.filter(x=>x.id!==r.id));notify('已删除「'+r.header+'」','red')}}>删除<//><//><//><//><//>`)}<//><//><//>
  <div className="tf"><span>已选 ${sel.length} / ${rows.length} 行</span><${Group} gap="lg"><${Group} gap="xs"><${Text} size="sm" fw=${500} c="dark">每页行数<//><${Select} size="xs" w=${72} value=${ps} onChange=${v=>{setPs(v);setPg(1)}} data=${['10','20','30']} allowDeselect=${false}/><//><${Pagination} size="sm" total=${pages} value=${pg} onChange=${setPg} withEdges/><//></div><//>`}
 </div></div></div>
 <${Modal} opened=${o} onClose=${()=>setO(false)} title="新建章节" centered><${TextInput} label="章节标题" placeholder="例如：项目背景" value=${nt} onChange=${e=>setNt(e.currentTarget.value)} data-autofocus/><${Group} justify="flex-end" mt="lg" gap="xs"><${Button} variant="default" onClick=${()=>setO(false)}>取消<//><${Button} onClick=${create}>创建<//><//><//><//>`}
createRoot(document.getElementById('app')).render(html`<${MantineProvider}><${Notifications} position="top-center"/><${Dash}/><//>`);
